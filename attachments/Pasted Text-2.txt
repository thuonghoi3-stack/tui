f_clamp(float value, float minValue, float maxValue) =>
    math.min(math.max(value, minValue), maxValue)

// Luminance-based contrast: dark ink on bright fills, light ink on dark fills.
f_text_on(color bg) =>
    float lum = 0.299 * color.r(bg) + 0.587 * color.g(bg) + 0.114 * color.b(bg)
    lum > 145 ? color.rgb(17, 24, 39) : headerText

// Vertical anti-collision: nudge a new label away from nearby existing labels.
f_avoid_overlap(array<label> labels, float desiredY, bool placeAbove, float safe) =>
    float y = desiredY
    float minGap = safe * 1.2
    int n = array.size(labels)
    if n > 0
        bool moved = true
        int guard = 0
        while moved and guard < 18
            moved := false
            for i = 0 to n - 1
                label lb = array.get(labels, i)
                if math.abs(label.get_x(lb) - bar_index) <= 26
                    if math.abs(label.get_y(lb) - y) < minGap
                        y := placeAbove ? y + minGap : y - minGap
                        moved := true
            guard += 1
    y

f_norm_high_good(float value, float weak, float strong) =>
    float span = strong - weak
    span == 0.0 ? 0.0 : f_clamp((value - weak) / span * 100.0, 0.0, 100.0)

f_norm_low_good(float value, float good, float bad) =>
    float span = bad - good
    span == 0.0 ? 0.0 : f_clamp((bad - value) / span * 100.0, 0.0, 100.0)

f_peak_score(float value, float lowValue, float idealValue, float highValue) =>
    float result = 0.0
    if value <= idealValue
        result := f_norm_high_good(value, lowValue, idealValue)
    else
        result := f_norm_low_good(value, idealValue, highValue)
    result

    float p = math.abs(price)
    p >= 10000.0 ? 1000.0 :
     p >= 1000.0 ? 100.0 :
     p >= 100.0 ? 10.0 :
     p >= 10.0 ? 1.0 :
     p >= 1.0 ? 0.10 :
     p >= 0.10 ? 0.01 :
     p >= 0.01 ? 0.001 : 0.0001

float atr = ta.atr(atrLen)
float safeAtr = math.max(nz(atr, ta.tr(true)), syminfo.mintick)
float atrBase = ta.sma(atr, atrBaseLen)
float atrRatio = safeAtr / math.max(nz(atrBase, safeAtr), syminfo.mintick)
float trendFast = ta.ema(close, trendLen)
float trendSlow = ta.ema(close, trendLen * 2)
float rangeHigh = ta.highest(high, obstructionLookback)
float rangeLow = ta.lowest(low, obstructionLookback)
float rangeMid = (rangeHigh + rangeLow) * 0.5
float rangeProxy = math.max(ta.highest(high, 90) - ta.lowest(low, 90), safeAtr * 8.0)

var float lastPivotHigh = na
var float lastPivotLow = na
var int lastPivotHighBar = na
var int lastPivotLowBar = na

float pivotHigh = ta.pivothigh(high, pivotLen, pivotLen)
float pivotLow = ta.pivotlow(low, pivotLen, pivotLen)

if not na(pivotHigh)
    lastPivotHigh := pivotHigh
    lastPivotHighBar := bar_index - pivotLen

if not na(pivotLow)
    lastPivotLow := pivotLow
    lastPivotLowBar := bar_index - pivotLen

bool havePivots = not na(lastPivotHigh) and not na(lastPivotLow)
int pivotContextDir = havePivots ? (close >= (lastPivotHigh + lastPivotLow) * 0.5 ? 1 : -1) : (close >= trendFast ? 1 : -1)
int directionOverride = planDirection == "Long Plan" ? 1 : planDirection == "Short Plan" ? -1 : 0
int autoDirection = planDirection == "Auto" ? pivotContextDir : directionOverride

float entryPx = na
float swingPx = na
float invalidPx = na
int planDir = na

if anchorMode == "Manual Price"
    bool manualDirReady = manualEntry > 0 and manualSwing > 0
    planDir := directionOverride != 0 ? directionOverride : manualDirReady ? (manualEntry >= manualSwing ? 1 : -1) : na
    entryPx := manualEntry > 0 ? manualEntry : na
    swingPx := manualSwing > 0 ? manualSwing : na
    invalidPx := manualStop > 0 ? manualStop : na
else if anchorMode == "Auto - Pivot High/Low"
    planDir := autoDirection
    entryPx := close
    swingPx := planDir == 1 ? lastPivotLow : lastPivotHigh
else
    planDir := directionOverride != 0 ? directionOverride : close >= rangeMid ? 1 : -1
    entryPx := close
    swingPx := planDir == 1 ? rangeLow : rangeHigh

if anchorMode != "Manual Price" and not na(entryPx) and not na(swingPx) and not na(planDir)
    if stopMode == "ATR Multiple"
        invalidPx := planDir == 1 ? entryPx - safeAtr * atrMult : entryPx + safeAtr * atrMult
    else if stopMode == "Swing Point"
        invalidPx := swingPx
    else
        invalidPx := planDir == 1 ? entryPx * (1.0 - stopPct / 100.0) : entryPx * (1.0 + stopPct / 100.0)

bool validBase = not na(entryPx) and not na(swingPx) and not na(invalidPx) and not na(planDir) and entryPx > 0 and math.abs(entryPx - invalidPx) > syminfo.mintick
float legSize = validBase ? math.abs(entryPx - swingPx) : na
float riskSize = validBase ? math.abs(entryPx - invalidPx) : na
float dirMult = planDir == 1 ? 1.0 : -1.0

f_compute_tp(int tier) =>
    float priceOut = na
    if validBase
        float fibTarget = entryPx + dirMult * legSize * math.max(f_get_fib(tier) - 1.0, 0.0)
        float rTarget = entryPx + dirMult * riskSize * f_get_r(tier)
        if calcMode == "Fibonacci Extensions"
            priceOut := fibTarget
        else if calcMode == "R-Multiples"
            priceOut := rTarget
        else
            priceOut := (fibTarget + rTarget) * 0.5
        if priceOut <= syminfo.mintick
            priceOut := na
    priceOut

float recentHighObstacle = ta.highest(high[1], obstructionLookback)
float recentLowObstacle = ta.lowest(low[1], obstructionLookback)

f_confluence(float targetPx) =>
    int score = 1
    if not na(targetPx)
        float unit = f_round_unit(targetPx)
        float rounded = math.round(targetPx / unit) * unit
        if math.abs(targetPx - rounded) <= safeAtr * 0.18
            score += 1
        float nearSwing = planDir == 1 ? recentHighObstacle : recentLowObstacle
        if not na(nearSwing) and math.abs(targetPx - nearSwing) <= safeAtr * 0.35
            score += 1
    math.min(score, 3)

var array<float> tpPx = array.from(float(na), float(na), float(na), float(na), float(na), float(na), float(na))
